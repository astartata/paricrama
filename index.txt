function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('🕉 Парикрама 2026')

    .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no');

function SS() { return SpreadsheetApp.getActiveSpreadsheet(); }

function safe_(v) {
  if (v === null || v === undefined) return '';
  if (v instanceof Date) {
    try { return Utilities.formatDate(v, Session.getScriptTimeZone(), 'dd.MM.yyyy HH:mm'); } catch(e) { return ''; }
  }
  return String(v);
}

function findCol_(hdr, substr) {
  var s = substr.toLowerCase();
  for (var j = 0; j < hdr.length; j++) { if (String(hdr[j]).toLowerCase().indexOf(s) >= 0) return j; }
  return -1;
}

// ═══ ВСЕ ДАННЫЕ ═══
function getAllData() {
  var guests = getGuests_();
  var rooms = getRooms_();
  var payments = getPayments_();
  var children = getChildren_();
  var team = getTeamOrVip_('Команда');
  var vip = getTeamOrVip_('Бюджетники');

  // Auto-sync team/VIP to hotel rooms
  try { syncTeamToHotels(); rooms = getRooms_(); } catch(e) {}
  // Auto-fix receipt access (max once per hour)
  try { var lf=PropertiesService.getScriptProperties().getProperty('lastFixReceipt')||'0';if(Date.now()-Number(lf)>3600000){fixReceiptAccess();PropertiesService.getScriptProperties().setProperty('lastFixReceipt',String(Date.now()));}} catch(e) {}

  // Статистика
  var gF=0, gM=0, noRoom=0, tariffs={}, paid=0, unpaid=0, refusals=0;
  var familyCount=0;
  var paidNames = {};
  for (var p = 0; p < payments.length; p++) {
    var names = payments[p].name.split(',');
    for (var n = 0; n < names.length; n++) paidNames[names[n].trim().toLowerCase()] = true;
  }
  for (var i = 0; i < guests.length; i++) {
    var g = guests[i];
    if (g.refusal) { refusals++; continue; }
    if (g.gender.indexOf('Матаджи') >= 0) gF++; else gM++;
    if (!g.assigned) noRoom++;
    tariffs[g.tariffShort || 'Без тарифа'] = (tariffs[g.tariffShort || 'Без тарифа'] || 0) + 1;
    if (paidNames[g.name.toLowerCase()]) paid++; else unpaid++;
    if (g.kids5 > 0 || g.kids14 > 0) familyCount++;
  }

  var tB=0, oB=0, hs={};
  for (var i = 0; i < rooms.length; i++) {
    var r = rooms[i];
    if (!hs[r.hotel]) hs[r.hotel] = {total:0, occ:0, rooms:0};
    hs[r.hotel].rooms++; hs[r.hotel].total += r.beds; hs[r.hotel].occ += r.occupied;
    tB += r.beds; oB += r.occupied;
  }

  var rub=0, eur=0;
  for (var i = 0; i < payments.length; i++) {
    if (payments[i].currency.indexOf('Рубл') >= 0) rub += payments[i].amount; else eur += payments[i].amount;
  }

  var services = {};
  for (var i = 0; i < guests.length; i++) {
    if (guests[i].refusal) continue;
    var svc = guests[i].service;
    if (svc) {
      var parts = svc.split(',');
      for (var j = 0; j < parts.length; j++) {
        var key = parts[j].trim();
        if (key) {
          if (!services[key]) services[key] = [];
          services[key].push(guests[i].name + '|' + (guests[i].spirit || '') + '|' + (guests[i].gender || ''));
        }
      }
    }
  }

  var tariffList = getTariffs_();
  var noRoomGuests = getNoRoomGuests_();
  var messages = getMessages_();
  
  // Read exchange rate and form ID from Настройки
  var rubRate = 100;
  var formId = '';
  try {
    var shN = SS().getSheetByName('Настройки');
    if (shN) {
      var nData = shN.getDataRange().getValues();
      for (var i = 0; i < nData.length; i++) {
        var key = String(nData[i][0]||'').toLowerCase();
        if (key.indexOf('курс') >= 0) rubRate = Number(nData[i][1]) || 100;
        if (key.indexOf('форма_анкеты') >= 0) formId = String(nData[i][1]||'').trim();
      }
    }
  } catch(e) {}

  return {
    guests: guests, rooms: rooms, payments: payments,
    children: children, team: team, vip: vip, services: services,
    tariffList: tariffList, noRoomGuests: noRoomGuests, messages: messages, rubRate: rubRate, formIdSet: !!formId,
    dash: {
      total: guests.length - refusals, gF: gF, gM: gM, noRoom: noRoom,
      paid: paid, unpaid: unpaid, refusals: refusals,
      familyCount: familyCount,
      tB: tB, oB: oB, fB: tB - oB,
      pct: tB ? Math.round(oB / tB * 100) : 0,
      hs: hs, tariffs: tariffs,
      payCnt: payments.length, rub: rub, eur: eur,
      teamCnt: team.length, vipCnt: vip.length, childCnt: children.length
    }
  };
}

// ── Отели (по заголовкам) ──
function getAssignMap_() {
  var sh = SS().getSheetByName('Отели');
  if (!sh) return {};
  var data = sh.getDataRange().getValues();
  if (data.length < 2) return {};
  var hdr = data[0];
  var cRoom = findCol_(hdr, 'Номер');
  var cG1 = findCol_(hdr, 'Гость 1');
  if (cG1 < 0) cG1 = 7;
  var map = {};
  for (var i = 1; i < data.length; i++) {
    var room = String(data[i][cRoom] || '').trim();
    if (!room) continue;
    for (var g = cG1; g <= cG1 + 3; g++) {
      var gn = String(data[i][g] || '').trim();
      if (gn && gn !== 'нет') map[gn.toLowerCase()] = room;
    }
  }
  return map;
}

// ── Дети ──
function getChildren_() {
  var sh = SS().getSheetByName('Дети');
  if (!sh) return [];
  var data = sh.getDataRange().getValues();
  if (data.length < 2) return [];
  var h = data[0], fc = function(s) { return findCol_(h, s); };
  var cParent=fc('Фамилия и Имя'), cChild=fc('Фамилия и Имя ребёнка'),
      cAge=fc('Возраст'), cTariff=fc('Тариф'), cRoom=fc('Номер'), cPaid=fc('Оплата подтверж');
  // Get parent tariffs from Ответы
  var parentTariffs = {};
  var shA = SS().getSheetByName('Ответы');
  if (shA) {
    var dA = shA.getDataRange().getValues();
    var hA = dA[0], cN = findCol_(hA, 'Фамилия и Имя'), cT = findCol_(hA, 'Тариф');
    if (cN >= 0 && cT >= 0) {
      for (var j = 1; j < dA.length; j++) {
        var pn = String(dA[j][cN]||'').trim().toLowerCase();
        if (pn) parentTariffs[pn] = String(dA[j][cT]||'').split('(')[0].trim();
      }
    }
  }
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    var r = data[i];
    var parent = (cParent >= 0) ? String(r[cParent]||'').trim() : '';
    var child = (cChild >= 0) ? String(r[cChild]||'').trim() : '';
    if (!parent && !child) continue;
    var tariff = (cTariff >= 0) ? String(r[cTariff]||'').trim() : '';
    // Auto-fill tariff from parent if empty
    if (!tariff && parent) {
      tariff = parentTariffs[parent.toLowerCase()] || '';
      if (tariff && cTariff >= 0) sh.getRange(i+1, cTariff+1).setValue(tariff);
    }
    rows.push({ parent:parent, childName:child, age:(cAge>=0)?(Number(r[cAge])||0):0,
      tariff:tariff, room:(cRoom>=0)?String(r[cRoom]||''):'',
      paid:(cPaid>=0)?String(r[cPaid]||'').trim():'' });
  }
  return rows;
}

function writeGuestCol_(guestName, colSubstr, value) {
  var sh = SS().getSheetByName('Ответы');
  if (!sh) return {ok:false, msg:'Лист не найден'};
  var data = sh.getDataRange().getValues();
  var hdr = data[0];
  var cName = findCol_(hdr, 'Фамилия и Имя');
  var cTarget = findCol_(hdr, colSubstr);
  if (cTarget < 0) return {ok:false, msg:'Столбец "'+colSubstr+'" не найден'};
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][cName]||'').trim().toLowerCase() === guestName.toLowerCase()) {
      sh.getRange(i+1, cTarget+1).setValue(value);
      return {ok:true};
    }
  }
  return {ok:false, msg:'Участник не найден'};
}

function blockRoom(roomId) {
  var sh = SS().getSheetByName('Отели');
  if (!sh) return {ok:false, msg:'Лист не найден'};
  var data = sh.getDataRange().getValues();
  var hdr = data[0], cRoom = findCol_(hdr, 'Номер'), cStatus = findCol_(hdr, 'Статус');
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][cRoom]).trim() === roomId) {
      sh.getRange(i+1, cStatus+1).setValue('Заблокирован');
      logAction_('Блокировка', roomId);
      return {ok:true};
    }
  }
  return {ok:false, msg:'Не найден'};
}

function unblockRoom(roomId) {
  var sh = SS().getSheetByName('Отели');
  if (!sh) return {ok:false, msg:'Лист не найден'};
  var data = sh.getDataRange().getValues();
  var hdr = data[0], cRoom = findCol_(hdr, 'Номер'), cStatus = findCol_(hdr, 'Статус'),
      cG1 = findCol_(hdr, 'Гость 1'), cBeds = findCol_(hdr, 'Места');
  if (cG1 < 0) cG1 = 7;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][cRoom]).trim() === roomId) {
      var occ = 0;
      for (var g = cG1; g <= cG1+3; g++) {
        var v = String(data[i][g]||'').trim();
        if (v && v !== 'нет') occ++;
      }
      var beds = Number(data[i][cBeds])||0;
      sh.getRange(i+1, cStatus+1).setValue(occ === 0 ? 'Свободен' : (occ >= beds ? 'Занят' : 'Частично'));
      logAction_('Разблокировка', roomId);
      return {ok:true};
    }
  }
  return {ok:false, msg:'Не найден'};
}

// ── Расселение ──
function assignGuestToRoom(guestName, roomId, skipFormUpdate) {
  var sh = SS().getSheetByName('Отели');
  if (!sh) return {ok:false, msg:'Лист не найден'};
  var data = sh.getDataRange().getValues();
  var hdr = data[0];
  var cRoom = findCol_(hdr, 'Номер');
  var cG1 = findCol_(hdr, 'Гость 1'); if (cG1<0) cG1=7;
  var cBeds = findCol_(hdr, 'Места');
  var cStatus = findCol_(hdr, 'Статус');
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][cRoom]).trim() === roomId) {
      for (var g = cG1; g <= cG1+3; g++) {
        var val = String(data[i][g]||'').trim();
        if (!val || val === 'нет') {
          sh.getRange(i+1, g+1).setValue(guestName);
          var beds = Number(data[i][cBeds])||0, occ = 0;
          for (var k=cG1; k<=cG1+3; k++) {
            var v2 = (k===g) ? guestName : String(data[i][k]||'').trim();
            if (v2 && v2 !== 'нет') occ++;
          }
          sh.getRange(i+1, cStatus+1).setValue(occ >= beds ? 'Занят' : 'Частично');
          logAction_('Бронь', guestName+' → '+roomId);
          autoFillChildRoom_(guestName, roomId);
          // Check if whole room — mark remaining slots as reserved
          try {
            var shA = SS().getSheetByName('Ответы');
            if (shA) {
              var aData = shA.getDataRange().getValues();
              var aHdr = aData[0];
              var cNameA = findCol_(aHdr, 'Фамилия и Имя');
              var cFamily = findCol_(aHdr, 'Состав');
              var cRoommate = findCol_(aHdr, 'Сосед');
              for (var a = 1; a < aData.length; a++) {
                if (String(aData[a][cNameA]||'').trim().toLowerCase() === guestName.toLowerCase()) {
                  var fam = String(aData[a][cFamily]||'').toLowerCase();
                  var rm = String(aData[a][cRoommate]||'').trim().toLowerCase();
                  var isWhole = fam.indexOf('без подселения') >= 0 || fam.indexOf('оплачиваю весь') >= 0 || rm === guestName.toLowerCase();
                  if (isWhole) {
                    // Fill remaining slots with "—" and set status Занят
                    var freshRow = sh.getRange(i+1, 1, 1, sh.getLastColumn()).getValues()[0];
                    for (var fg = cG1; fg <= cG1+3; fg++) {
                      var fv = String(freshRow[fg]||'').trim();
                      if (!fv || fv === 'нет') sh.getRange(i+1, fg+1).setValue('—');
                    }
                    sh.getRange(i+1, cStatus+1).setValue('Занят');
                    logAction_('Весь номер', guestName+' → '+roomId);
                  }
                  break;
                }
              }
            }
          } catch(e) {}
          // Re-read room data after children added
          var updatedRow = sh.getRange(i+1, 1, 1, sh.getLastColumn()).getValues()[0];
          sendBookingEmail_(guestName, roomId, updatedRow, hdr);
          if (!skipFormUpdate) { try { var ufr=updateFormTariffs(); logAction_('Форма авто', ufr.msg||''); } catch(e) { logAction_('Форма ошибка', String(e)); } }
          return {ok:true, msg:guestName+' → '+roomId};
        }
      }
      return {ok:false, msg:'Номер полностью занят'};
    }
  }
  return {ok:false, msg:'Номер не найден'};
}

function removeChildrenFromRoom_(parentName, roomId) {
  var shD = SS().getSheetByName('Дети');
  if (!shD) return;
  var dData = shD.getDataRange().getValues();
  var dHdr = dData[0];
  var cParent = findCol_(dHdr, 'Фамилия и Имя');
  var cChild = findCol_(dHdr, 'Фамилия и Имя ребёнка');
  var cRoom = findCol_(dHdr, 'Номер');
  var cAge = findCol_(dHdr, 'Возраст');
  if (cParent < 0 || cRoom < 0) return;
  
  var childNames = [];
  var kids5 = 0;
  for (var i = 1; i < dData.length; i++) {
    if (String(dData[i][cParent]||'').trim().toLowerCase() === parentName.toLowerCase()) {
      shD.getRange(i+1, cRoom+1).setValue('');
      var age = Number(dData[i][cAge]) || 0;
      if (age >= 5) childNames.push(String(dData[i][cChild]||'').trim());
      else kids5++;
    }
  }
  
  if (childNames.length > 0 || kids5 > 0) {
    var shH = SS().getSheetByName('Отели');
    if (!shH) return;
    var hData = shH.getDataRange().getValues();
    var hdr = hData[0];
    var cRoomH = findCol_(hdr, 'Номер');
    var cG1 = findCol_(hdr, 'Гость 1'); if (cG1 < 0) cG1 = 7;
    var cBeds = findCol_(hdr, 'Места');
    var cStatus = findCol_(hdr, 'Статус');
    var cKids5 = findCol_(hdr, 'Дети до 5');
    
    for (var r = 1; r < hData.length; r++) {
      if (String(hData[r][cRoomH]).trim() === roomId) {
        // Remove children 5-14 from guest slots
        for (var ci = 0; ci < childNames.length; ci++) {
          for (var g = cG1; g <= cG1+3; g++) {
            if (String(hData[r][g]||'').trim().toLowerCase() === childNames[ci].toLowerCase()) {
              shH.getRange(r+1, g+1).setValue('');
              break;
            }
          }
        }
        // Remove kids under 5
        if (kids5 > 0 && cKids5 >= 0) {
          var existing = Number(hData[r][cKids5]) || 0;
          shH.getRange(r+1, cKids5+1).setValue(Math.max(0, existing - kids5));
        }
        // Recalculate occupancy
        var beds = Number(hData[r][cBeds]) || 0;
        var occ = 0;
        for (var k = cG1; k <= cG1+3; k++) {
          var v2 = String(shH.getRange(r+1, k+1).getValue()||'').trim();
          if (v2 && v2 !== 'нет') occ++;
        }
        shH.getRange(r+1, cStatus+1).setValue(occ === 0 ? 'Свободен' : (occ >= beds ? 'Занят' : 'Частично'));
        break;
      }
    }
  }
}

function autoFillChildRoom_(parentName, roomId) {
  try {
    var shD = SS().getSheetByName('Дети');
    if (!shD) return;
    var dData = shD.getDataRange().getValues();
    var dHdr = dData[0];
    var cParent = findCol_(dHdr, 'Фамилия и Имя');
    var cChild = findCol_(dHdr, 'Фамилия и Имя ребёнка');
    var cRoom = findCol_(dHdr, 'Номер');
    var cAge = findCol_(dHdr, 'Возраст');
    if (cParent < 0 || cRoom < 0) return;
    
    var kids5 = 0;
    var kidsOver5 = [];
    
    for (var i = 1; i < dData.length; i++) {
      if (String(dData[i][cParent]||'').trim().toLowerCase() === parentName.toLowerCase()) {
        shD.getRange(i+1, cRoom+1).setValue(roomId);
        var age = Number(dData[i][cAge]) || 0;
        if (age >= 5) {
          kidsOver5.push(String(dData[i][cChild]||'').trim());
        } else {
          kids5++;
        }
      }
    }
    
    // Book children 5-14 into guest slots (they occupy a bed)
    if (kidsOver5.length > 0) {
      var shH = SS().getSheetByName('Отели');
      if (!shH) return;
      var hData = shH.getDataRange().getValues();
      var hdr = hData[0];
      var cRoomH = findCol_(hdr, 'Номер');
      var cG1 = findCol_(hdr, 'Гость 1'); if (cG1 < 0) cG1 = 7;
      var cBeds = findCol_(hdr, 'Места');
      var cStatus = findCol_(hdr, 'Статус');
      var cKids5 = findCol_(hdr, 'Дети до 5');
      
      for (var r = 1; r < hData.length; r++) {
        if (String(hData[r][cRoomH]).trim() === roomId) {
          // Write each child 5-14 to next free guest slot
          var usedSlots = {};
          for (var ki = 0; ki < kidsOver5.length; ki++) {
            for (var g = cG1; g <= cG1+3; g++) {
              if (usedSlots[g]) continue;
              var val = String(hData[r][g]||'').trim();
              if (!val || val === 'нет') {
                shH.getRange(r+1, g+1).setValue(kidsOver5[ki]);
                hData[r][g] = kidsOver5[ki]; // update local cache
                usedSlots[g] = true;
                break;
              }
            }
          }
          // Write kids under 5 count
          if (kids5 > 0 && cKids5 >= 0) {
            var existing = Number(hData[r][cKids5]) || 0;
            shH.getRange(r+1, cKids5+1).setValue(existing + kids5);
          }
          // Recalculate occupancy
          var beds = Number(hData[r][cBeds]) || 0;
          var occ = 0;
          for (var k = cG1; k <= cG1+3; k++) {
            var v2 = String(shH.getRange(r+1, k+1).getValue()||'').trim();
            if (v2 && v2 !== 'нет') occ++;
          }
          shH.getRange(r+1, cStatus+1).setValue(occ >= beds ? 'Занят' : 'Частично');
          break;
        }
      }
    }
  } catch(e) {}
}

function syncTeamToHotels() {
  var sh = SS().getSheetByName('Отели');
  if (!sh) return {ok:false, msg:'Лист Отели не найден'};
  var data = sh.getDataRange().getValues();
  var hdr = data[0];
  var cRoom = findCol_(hdr, 'Номер');
  var cG1 = findCol_(hdr, 'Гость 1'); if (cG1<0) cG1=7;
  var cBeds = findCol_(hdr, 'Места');
  var cStatus = findCol_(hdr, 'Статус');
  
  // Collect all currently placed guests
  var placed = {};
  for (var i = 1; i < data.length; i++) {
    for (var g = cG1; g <= cG1+3; g++) {
      var v = String(data[i][g]||'').trim();
      if (v && v !== 'нет') placed[v.toLowerCase()] = true;
    }
  }
  
  // Read Команда + Бюджетники
  var added = 0;
  var sheets = ['Команда', 'Бюджетники'];
  for (var s = 0; s < sheets.length; s++) {
    var src = SS().getSheetByName(sheets[s]);
    if (!src) continue;
    var srcData = src.getDataRange().getValues();
    for (var j = 1; j < srcData.length; j++) {
      var name = String(srcData[j][3]||srcData[j][4]||'').trim(); // col 3=name, 4=spirit
      var roomId = String(srcData[j][2]||'').trim(); // col 2=roomId
      if (!name || !roomId) continue;
      if (placed[name.toLowerCase()]) continue; // already placed
      
      // Find room in Отели and add guest
      for (var i = 1; i < data.length; i++) {
        if (String(data[i][cRoom]).trim() === roomId) {
          for (var g = cG1; g <= cG1+3; g++) {
            var val = String(data[i][g]||'').trim();
            if (!val || val === 'нет') {
              sh.getRange(i+1, g+1).setValue(name);
              data[i][g] = name; // update local
              placed[name.toLowerCase()] = true;
              // Update status
              var occ = 0;
              for (var k=cG1; k<=cG1+3; k++) {
                var v2 = String(data[i][k]||'').trim();
                if (v2 && v2 !== 'нет') occ++;
              }
              var beds = Number(data[i][cBeds])||0;
              sh.getRange(i+1, cStatus+1).setValue(occ >= beds ? 'Занят' : 'Частично');
              added++;
              break;
            }
          }
          break;
        }
      }
    }
  }
  logAction_('Синхр. команды', added + ' добавлено');
  return {ok:true, msg:'Добавлено: ' + added};
}

function updateFormTariffs() {
  var shN = SS().getSheetByName('Настройки');
  if (!shN) return {ok:false, msg:'Настройки не найдены'};
  var nData = shN.getDataRange().getValues();
  var formId = '';
  for (var i = 0; i < nData.length; i++) {
    if (String(nData[i][0]||'').toLowerCase().indexOf('форма_анкеты') >= 0) {
      formId = String(nData[i][1]||'').trim(); break;
    }
  }
  if (!formId) return {ok:false, msg:'ID формы не найден в Настройках'};
  
  // Count free beds per tariff from Отели
  var shH = SS().getSheetByName('Отели');
  if (!shH) return {ok:false, msg:'Лист Отели не найден'};
  var hData = shH.getDataRange().getValues();
  var hdr = hData[0];
  var cTariff = findCol_(hdr, 'Тариф');
  var cBeds = findCol_(hdr, 'Места');
  var cG1 = findCol_(hdr, 'Гость 1'); if (cG1 < 0) cG1 = 7;
  var cStatus = findCol_(hdr, 'Статус');
  
  var tariffFree = {};
  for (var i = 1; i < hData.length; i++) {
    var t = String(hData[i][cTariff]||'').trim();
    var st = String(hData[i][cStatus]||'');
    if (!t || st === 'Заблокирован') continue;
    var beds = Number(hData[i][cBeds]) || 0;
    var occ = 0;
    for (var g = cG1; g <= cG1 + 3; g++) {
      var v = String(hData[i][g]||'').trim();
      if (v && v !== 'нет') occ++;
    }
    if (!tariffFree[t]) tariffFree[t] = 0;
    tariffFree[t] += Math.max(0, beds - occ);
  }
  
  // Open form
  var form;
  try { form = FormApp.openById(formId); }
  catch(e) { return {ok:false, msg:'Не удалось открыть форму: ' + e.message}; }
  
  var items = form.getItems();
  var tariffItem = null;
  for (var i = 0; i < items.length; i++) {
    if (String(items[i].getTitle()).toLowerCase().indexOf('тариф') >= 0) {
      tariffItem = items[i]; break;
    }
  }
  if (!tariffItem) return {ok:false, msg:'Вопрос "Тариф" не найден в форме'};
  
  var listItem;
  try { listItem = tariffItem.asListItem(); } catch(e) {
    try { listItem = tariffItem.asMultipleChoiceItem(); } catch(e2) {
      return {ok:false, msg:'Вопрос тарифа не является списком'};
    }
  }
  
  // Save original choices on first run
  var props = PropertiesService.getScriptProperties();
  var savedKey = 'originalTariffChoices';
  var savedJson = props.getProperty(savedKey);
  var masterChoices;
  
  if (!savedJson) {
    // First run — save current form choices as master
    masterChoices = listItem.getChoices().map(function(ch) { return ch.getValue(); });
    props.setProperty(savedKey, JSON.stringify(masterChoices));
  } else {
    masterChoices = JSON.parse(savedJson);
  }
  
  // Build new choices from master: show if has free beds or not a room tariff
  var newChoices = [];
  var hidden = [];
  for (var c = 0; c < masterChoices.length; c++) {
    var text = masterChoices[c];
    var isRoomTariff = false;
    var hasFree = false;
    for (var t in tariffFree) {
      if (text.toLowerCase().indexOf(t.toLowerCase()) >= 0) {
        isRoomTariff = true;
        if (tariffFree[t] > 0) hasFree = true;
        break;
      }
    }
    if (!isRoomTariff || hasFree) {
      newChoices.push(listItem.createChoice(text));
    } else {
      hidden.push(text);
    }
  }
  
  if (newChoices.length === 0) {
    return {ok:false, msg:'Все тарифы заполнены!'};
  }
  
  listItem.setChoices(newChoices);
  logAction_('Форма обновлена', hidden.length + ' скрыто, ' + newChoices.length + ' доступно');
  return {ok:true, msg:'Доступно: ' + newChoices.length + ', скрыто: ' + hidden.length + (hidden.length ? '\n' + hidden.join('\n') : '')};
}

  for (var i = 1; i < data.length; i++) {
    var url = String(data[i][cReceipt]||'').trim();
    if (!url) continue;
    try {
      var m = url.match(/[?&]id=([a-zA-Z0-9_-]+)/);
      if (!m) m = url.match(/\/d\/([a-zA-Z0-9_-]+)/);
      if (m) {
        var file = DriveApp.getFileById(m[1]);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        fixed++;
      }
    } catch(e) {}
  }
  logAction_('Доступ к квитанциям', fixed + ' файлов открыто');
  return {ok:true, msg:'Открыт доступ к ' + fixed + ' квитанциям'};
}

// ── Уведомление о новой оплате ──
function onPaymentSubmit(e) {
  try {
    var shN = SS().getSheetByName('Настройки');
    if (!shN) return;
    var nData = shN.getDataRange().getValues();
    var emails = '';
    for (var i = 0; i < nData.length; i++) {
      if (String(nData[i][0]||'').toLowerCase().indexOf('email_ответственных_оплаты') >= 0) {
        emails = String(nData[i][1]||'').trim();
        break;
      }
    }
    if (!emails) return;
    var sh = SS().getSheetByName('Оплаты');
    if (!sh) return;
    var last = sh.getLastRow();
    var data = sh.getRange(last, 1, 1, sh.getLastColumn()).getValues()[0];
    var hdr = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
    var cName = -1, cAmount = -1, cCurr = -1;
    for (var j = 0; j < hdr.length; j++) {
      var h = String(hdr[j]).toLowerCase();
      if (h.indexOf('фамилия') >= 0 || h.indexOf('имя') >= 0) cName = j;
      if (h.indexOf('сумма') >= 0) cAmount = j;
      if (h.indexOf('валюта') >= 0) cCurr = j;
    }
    var name = cName >= 0 ? String(data[cName]||'') : 'Неизвестно';
    var amount = cAmount >= 0 ? String(data[cAmount]||'') : '?';
    var curr = cCurr >= 0 ? String(data[cCurr]||'') : '';
    var subject = '💰 Новая оплата: ' + name;
    var body = 'Получена оплата:\n\n';
    body += 'Участник: ' + name + '\n';
    body += 'Сумма: ' + amount + ' ' + curr + '\n';
    body += 'Дата: ' + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd.MM.yyyy HH:mm') + '\n\n';
    body += 'Подтвердите аванс в приложении.';
    var emailList = emails.split(',');
    for (var e = 0; e < emailList.length; e++) {
      var em = emailList[e].trim();
      if (em) MailApp.sendEmail(em, subject, body);
    }
  } catch(err) {}
}

// Установка триггера (запустить один раз вручную)
function setupPaymentTrigger() {
  // Remove old triggers
  var triggers = ScriptApp.getProjectTriggers();
  triggers.forEach(function(t) {
    if (t.getHandlerFunction() === 'onPaymentSubmit') ScriptApp.deleteTrigger(t);
  });
  // Add new
  ScriptApp.newTrigger('onPaymentSubmit')
    .forSpreadsheet(SS())
    .onFormSubmit()
    .create();
  return {ok: true, msg: 'Триггер установлен'};
}

function testFormTariffs() {
  // Diagnostic: show what's happening
  var shH = SS().getSheetByName('Отели');
  var shT = SS().getSheetByName('Тарифы');
  var diag = '';
  
  if (shH) {
    var hData = shH.getDataRange().getValues();
    var hdr = hData[0];
    var cTariff = findCol_(hdr, 'Тариф');
    var cBeds = findCol_(hdr, 'Места');
    var cG1 = findCol_(hdr, 'Гость 1'); if (cG1 < 0) cG1 = 7;
    var cStatus = findCol_(hdr, 'Статус');
    var tariffFree = {};
    for (var i = 1; i < hData.length; i++) {
      var t = String(hData[i][cTariff]||'').trim();
      if (!t) continue;
      var beds = Number(hData[i][cBeds]) || 0;
      var occ = 0;
      for (var g = cG1; g <= cG1+3; g++) {
        var v = String(hData[i][g]||'').trim();
        if (v && v !== 'нет') occ++;
      }
      if (!tariffFree[t]) tariffFree[t] = 0;
      tariffFree[t] += Math.max(0, beds - occ);
    }
    for (var t in tariffFree) {
      diag += t + ': ' + tariffFree[t] + ' св. мест\n';
    }
  }
  
  if (shT) {
    var tData = shT.getDataRange().getValues();
    var tHdr = tData[0];
    var cTName = findCol_(tHdr, 'Тариф');
    diag += '\nТарифы лист: ';
    if (cTName >= 0) {
      for (var i = 1; i < tData.length; i++) {
        diag += String(tData[i][cTName]||'') + '; ';
      }
    }
  }
  
  var result = updateFormTariffs();
  result.diag = diag;
  return result;
}
